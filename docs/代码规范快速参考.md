# Python 代码规范快速参考

## 📌 核心规范（必须遵守）

### 1. 所有代码必须在函数体内 ✅

```python
# ❌ 错误：全局作用域的裸代码
logger = get_app_logger()
config = load_config()
result = process(config)

# ✅ 正确：封装在函数中
def main() -> None:
    """主函数"""
    logger = get_app_logger()
    config = load_config()
    result = process(config)
    logger.info(f"结果: {result}")

if __name__ == "__main__":
    main()
```

### 2. 使用 logger 而非 print() ✅

```python
# ❌ 错误：使用 print
print("开始处理数据")
print(f"错误: {error}")

# ✅ 正确：使用 logger
from backend.utils.logger import get_app_logger

logger = get_app_logger()
logger.info("开始处理数据")
logger.error(f"处理失败: {error}", exc_info=True)
```

### 3. 完整的文档字符串 ✅

```python
# ❌ 错误：文档不完整
def process_data(data, config=None):
    """处理数据"""
    pass

# ✅ 正确：完整文档
def process_data(data: List[Dict], config: Optional[Dict] = None) -> bool:
    """
    处理数据并返回处理结果
    
    对输入数据进行验证、清洗和转换。
    
    Args:
        data: 待处理的数据列表，每项为字典
        config: 可选的配置字典，包含处理参数
        
    Returns:
        bool: 处理是否成功
        
    Raises:
        ValueError: 数据格式无效时
        
    Example:
        >>> data = [{"name": "test", "value": 1}]
        >>> result = process_data(data)
        >>> print(result)
        True
    """
    pass
```

### 4. 使用类型注解 ✅

```python
from typing import List, Dict, Optional, Tuple

# ❌ 错误：缺少类型注解
def calculate(a, b):
    return a + b

# ✅ 正确：有类型注解
def calculate(a: float, b: float) -> float:
    """计算两数之和"""
    return a + b

# 复杂类型
def process_users(
    users: List[Dict[str, Any]],
    filters: Optional[Dict[str, str]] = None
) -> Tuple[int, List[Dict]]:
    """处理用户列表"""
    pass
```

## 📚 文档字符串模板

### 函数文档字符串

```python
def function_name(param1: Type1, param2: Type2) -> ReturnType:
    """
    简短的一句话描述（必需）
    
    详细描述，说明函数的具体行为和注意事项（可选）。
    可以分多段说明。
    
    Args:
        param1: 参数1的说明
        param2: 参数2的说明
        
    Returns:
        ReturnType: 返回值的说明
        
    Raises:
        ExceptionType: 异常说明
        
    Example:
        >>> result = function_name(arg1, arg2)
        >>> print(result)
        expected_output
    """
    pass
```

### 类文档字符串

```python
class ClassName:
    """
    简短的一句话描述（必需）
    
    详细描述类的功能和用途（可选）。
    
    Attributes:
        attr1: 属性1的说明
        attr2: 属性2的说明
        
    Example:
        >>> obj = ClassName()
        >>> obj.method()
    """
    
    def __init__(self, param: Type) -> None:
        """
        初始化类实例
        
        Args:
            param: 参数说明
        """
        self.attr1: Type1 = value1
        self.attr2: Type2 = value2
```

### 模块文档字符串

```python
"""
模块名称

模块的详细描述，说明模块的主要功能和用途。
可以包含使用示例和注意事项。

Author: 作者名
Date: 创建日期
"""

import os
import sys
```

## 🎨 代码格式规范

### 导入顺序

```python
"""模块文档字符串"""

# 1. 标准库导入
import os
import sys
from pathlib import Path
from typing import List, Dict, Optional

# 2. 第三方库导入
import numpy as np
import pandas as pd
from fastapi import FastAPI

# 3. 本地模块导入
from backend.utils.logger import get_app_logger
from backend.models.user import User
```

### 命名规范

```python
# 模块名：小写字母，下划线分隔
# user_data_processor.py

# 类名：大驼峰命名
class UserDataProcessor:
    pass

# 函数名：小写字母，下划线分隔
def process_user_data():
    pass

# 常量：大写字母，下划线分隔
MAX_RETRY_COUNT = 3
DEFAULT_TIMEOUT = 30

# 变量名：小写字母，下划线分隔
user_count = 0
is_valid = True

# 私有成员：单下划线前缀
def _internal_helper():
    pass
```

### 空行规范

```python
"""模块文档字符串"""

import os  # 导入后空两行


CONSTANT_VALUE = 100  # 常量定义


class MyClass:  # 类定义前空两行
    """类文档字符串"""
    
    def __init__(self):  # 方法间空一行
        """初始化"""
        pass
    
    def method_one(self):  # 方法间空一行
        """方法一"""
        pass


def standalone_function():  # 顶级函数前空两行
    """独立函数"""
    pass
```

## 🔧 Logger 使用指南

### 获取 Logger

```python
from backend.utils.logger import (
    get_app_logger,    # 应用主日志
    get_asr_logger,    # 语音识别日志
    get_api_logger,    # API 日志
)

# 在类中使用
class MyService:
    def __init__(self):
        self.logger = get_app_logger()
        self.logger.info("服务初始化完成")

# 在函数中使用
def process_data():
    logger = get_app_logger()
    logger.info("开始处理数据")
```

### Logger 方法

```python
logger.debug("调试信息")           # 详细的调试信息
logger.info("信息日志")            # 一般的信息日志
logger.warning("警告信息")         # 警告信息
logger.error("错误信息")           # 错误信息
logger.error("错误", exc_info=True)  # 错误 + 堆栈跟踪
```

### 便捷函数

```python
from backend.utils.logger import (
    log_info,
    log_error,
    log_warning,
    log_debug
)

log_info("操作成功")
log_error("操作失败", exc_info=True)
log_warning("配置缺失")
log_debug("变量值: x")
```

## ⚠️ 异常处理

### 完整的异常处理

```python
def read_config_file(file_path: str) -> dict:
    """
    读取配置文件
    
    Args:
        file_path: 配置文件路径
        
    Returns:
        dict: 配置字典
        
    Raises:
        FileNotFoundError: 文件不存在
        json.JSONDecodeError: JSON 格式错误
    """
    logger = get_app_logger()
    
    try:
        with open(file_path, 'r', encoding='utf-8') as f:
            config = json.load(f)
        logger.info(f"成功加载配置文件: {file_path}")
        return config
        
    except FileNotFoundError:
        logger.error(f"配置文件不存在: {file_path}")
        raise
        
    except json.JSONDecodeError as e:
        logger.error(f"配置文件格式错误: {file_path}", exc_info=True)
        raise
        
    except Exception as e:
        logger.error(f"读取配置文件失败: {e}", exc_info=True)
        raise
```

### 资源清理

```python
def process_file(file_path: str) -> None:
    """处理文件"""
    logger = get_app_logger()
    file_handle = None
    
    try:
        file_handle = open(file_path, 'r')
        data = file_handle.read()
        process_data(data)
        logger.info(f"文件处理完成: {file_path}")
        
    except Exception as e:
        logger.error(f"文件处理失败: {e}", exc_info=True)
        raise
        
    finally:
        # 确保资源被释放
        if file_handle:
            file_handle.close()
            logger.debug("文件句柄已关闭")
```

## ✅ 检查清单

在提交代码前，使用此清单检查：

- [ ] 所有代码都在函数体内
- [ ] 使用 logger 工具类，没有 print()
- [ ] 每个函数都有完整的文档字符串
- [ ] 文档字符串包含 Args、Returns、Raises
- [ ] 复杂逻辑有行内注释
- [ ] 使用类型注解
- [ ] 遵循 PEP 8 命名规范
- [ ] 每行不超过 79 个字符
- [ ] 导入语句按标准库、第三方库、本地模块分组
- [ ] 异常处理完整，有日志记录
- [ ] 资源使用后正确释放

```bash

isort your_file.py

# 代码检查
flake8 your_file.py --max-line-length=100

# 类型检查
mypy your_file.py
```

```json

```
